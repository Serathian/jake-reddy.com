import { Font } from 'astro:assets';
import '../styles/global.css';

}

const { title, description } = Astro.props;
---

  <body class="min-h-screen bg-white text-black">
    <slot />
  </body>
</html>

- global.css import makes all design tokens available
- CSS variable names must match exactly from Plan 02
- Body classes demonstrate design system usage (bg-white, text-black)
